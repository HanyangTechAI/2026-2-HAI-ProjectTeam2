실효값 검증 - 기준: o-drive
python -c "import json, os, numpy as np, soundfile as sf; from pathlib import Path; from collections import defaultdict; out_dir = Path(os.environ['TEMP']) / 'pilot_wet_output'; rms_by_drive = defaultdict(list); [rms_by_drive[json.loads(p.read_text('utf-8'))['ground_truth_chain']['chain'][0]['effect_type']].append(float(np.sqrt(np.mean(sf.read(out_dir / 'audio' / f\"{json.loads(p.read_text('utf-8'))['target_id']}.wav\")[0]**2)))) for p in sorted((out_dir / 'labels').glob('*.json'))]; means = {d: np.mean(v) for d, v in rms_by_drive.items()}; target = float(means['o_drive']); print(f'기준 목표 RMS (o_drive): {target:.5f} ({20*np.log10(target):.2f} dBFS)\n'); [print(f'{d:16s} | 평균 RMS: {m:.5f} ({20*np.log10(m):6.2f} dBFS) | 기준 대비: {(m-target)/target*100:+6.1f}% ({20*np.log10(m/target):+5.2f} dB)') for d, m in sorted(means.items(), key=lambda x: x[1], reverse=True)]"
